// Request idioms this reader does NOT know (Phase 333.11.1, SPEC-1 §3.6),
// each reported with its line, one unread line each, and never read as
// something else. SPEC-1's eight: `target.append(…)`; `if since != nil`; a
// literal query value (`&keep=top`); an interpolation calling an unknown
// function; a `switch` in a builder; `try? c.encode(…)`; an `encode` of a
// computed expression; `DoorLine(keeps: flag)`. Beside them, one line for
// every other refusal the request reader makes, so each has a test: a String
// sent without `queryValue`, text around an interpolation, `max` of a
// variable, `if let` of a non-optional, `&` appended to a target with no `?`,
// an interpolation in the path, a static target with a query, a name sent
// twice, a `queryValue` that is not the builder's own, a `Double`, a `[T?]`, an
// object of fields inside a value, a type that is not Encodable, a
// synthesized enum, an `encode` of another signature, a `let` container, a
// key not in CodingKeys, a key written twice, a `default:`, `.rawValue` of a
// String, a type that holds itself, a `signedPost` case of another shape, an
// encoder that does not sort its keys, two encoders, and a kept line outside
// any type; and, under each of those rules, one line for each of its own
// refusals (`if var`, a signature, a default, an instance func, a value that
// is not a string, no return, an optional, the wrong kind, no CodingKeys, an
// enum encoder with no switch, two encoders of one type, a signedPost with
// no switch, a switch on something else, a body type it cannot find, a verb
// encoded twice). Each line the reader must refuse ends `// unread`.
let globalLine = DoorLine(keeps: true) // unread

enum Keep: String, Equatable, Sendable {
    case top, bottom
}

final class DoorClient {
    static let queryTarget = "/v1/l?x=1" // unread

    static func appendTarget(_ sessionId: String, to: Int?) -> String {
        var target = "/v1/a?id=\(queryValue(sessionId))"
        if let to { target.append("&to=\(to)") } // unread
        return target
    }

    static func nilTarget(_ sessionId: String, since: String?) -> String {
        var target = "/v1/b?id=\(queryValue(sessionId))"
        if since != nil { target += "&since=\(queryValue(since ?? ""))" } // unread
        return target
    }

    static func literalTarget(_ sessionId: String) -> String {
        "/v1/c?id=\(queryValue(sessionId))&keep=top" // unread
    }

    static func unknownTarget(_ sessionId: String) -> String {
        "/v1/d?id=\(escape(sessionId))" // unread
    }

    static func switchTarget(_ keep: Keep) -> String {
        var target = "/v1/e"
        switch keep { // unread
        case .top: target += "?keep=top"
        case .bottom: target += "?keep=bottom"
        }
        return target
    }

    static func bareTarget(_ sessionId: String) -> String {
        "/v1/f?id=\(sessionId)" // unread
    }

    static func aroundTarget(_ sessionId: String) -> String {
        "/v1/g?id=x\(queryValue(sessionId))" // unread
    }

    static func maxTarget(_ limit: Int, floor: Int) -> String {
        "/v1/i?limit=\(max(floor, limit))" // unread
    }

    static func unwrapTarget(_ sessionId: String) -> String {
        var target = "/v1/j?a=\(queryValue(sessionId))"
        if let sessionId { target += "&id=\(queryValue(sessionId))" } // unread
        return target
    }

    static func noQueryTarget(to: Int?) -> String {
        var target = "/v1/k"
        if let to { target += "&to=\(to)" } // unread
        return target
    }

    static func pathTarget(_ sessionId: String) -> String {
        "/v1/\(queryValue(sessionId))/x" // unread
    }

    static func twiceTarget(_ a: String, _ b: String) -> String {
        "/v1/m?id=\(queryValue(a))&id=\(queryValue(b))" // unread
    }

    static func notStringTarget(_ sessionId: String) -> Int { // unread
        1
    }

    static func defaultTarget(_ sessionId: String = "x") -> String { // unread
        "/v1/o?id=\(queryValue(sessionId))"
    }

    func instanceTarget(_ sessionId: String) -> String { // unread
        "/v1/o?id=\(Self.queryValue(sessionId))"
    }

    static func exprTarget(_ sessionId: String) -> String {
        prefix + "/v1/o" // unread
    }

    static func varNotStringTarget(_ sessionId: String) -> String {
        var target = prefix // unread
        return target
    }

    static func noReturnTarget(_ sessionId: String) -> String {
        var target = "/v1/o?id=\(queryValue(sessionId))"
        target // unread
    }

    static func questionTarget(_ sessionId: String, to: Int?) -> String {
        var target = "/v1/p?id=\(queryValue(sessionId))"
        if let to { target += "?to=\(to)" } // unread
        return target
    }

    static func emptyNameTarget(_ sessionId: String) -> String {
        "/v1/q?=\(queryValue(sessionId))" // unread
    }

    static func classPropTarget(_ holder: Holder) -> String {
        "/v1/r?id=\(queryValue(holder.name))" // unread
    }

    static func optionalPropTarget(_ query: Query?) -> String {
        "/v1/r?id=\(queryValue(query.name))" // unread
    }

    static func optionalTarget(_ sessionId: String?) -> String {
        "/v1/s?id=\(queryValue(sessionId))" // unread
    }

    static func rawStringTarget(_ sessionId: String) -> String {
        "/v1/t?id=\(sessionId.rawValue)" // unread
    }

    static func numberThroughTarget(_ count: Int) -> String {
        "/v1/u?n=\(queryValue(count))" // unread
    }

    static func boolTarget(_ flag: Bool) -> String {
        "/v1/v?flag=\(flag)" // unread
    }

    static func varBindTarget(_ sessionId: String, to: Int?) -> String {
        var target = "/v1/w?id=\(queryValue(sessionId))"
        if var to { target += "&to=\(to)" } // unread
        return target
    }

    static func queryValue(_ value: String) -> String {
        value
    }

    private func signedPost(route: WriteRoute) -> Data? {
        let body: Data
        do {
            let encoder = JSONEncoder()
            encoder.outputFormatting = [.sortedKeys]
            switch route {
            case .first(let session, let write):
                body = try encoder.encode(TryBody(session: session, write: write))
            case .made(let session): // unread
                body = Data(session.utf8)
            }
        } catch {
            return nil
        }
        return body
    }
}
