// The request reader's write bodies and kept lines (Phase 333.11.1, SPEC-1
// §3.6): a synthesized body of Bool, Int (renamed in CodingKeys), a raw enum,
// a `T?` the synthesized encoder leaves out (sometimes) and a String; a custom
// encoder writing a `String?` as null (always, null), an `encodeIfPresent`
// (sometimes) and an array of an enum item encoder of two cases, its keys in
// an unsorted order the record sorts; `signedPost`'s switch naming each body;
// and three types holding two, three and one `DoorLine(keeps: true)`, one
// `DoorLine(keeps: false)` beside them that is not a kept line, and the one
// in `Pane.Inner` counted against the innermost type. Nothing here is unread.
enum Keep: String, Equatable, Sendable {
    case top, bottom
}

enum KeyName: String, CaseIterable, Sendable {
    case escape = "Escape"
    case enter = "Enter"
}

struct PlainBody: Encodable, Sendable {
    let batch: Bool
    let count: Int
    let keep: Keep
    let note: String?
    let session: String

    enum CodingKeys: String, CodingKey {
        case batch, keep, note, session
        case count = "n"
    }
}

enum Item: Equatable, Sendable {
    case text(String)
    case key(KeyName)
}

extension Item: Encodable {
    enum CodingKeys: String, CodingKey {
        case typedText = "t"
        case keyName = "k"
    }

    func encode(to encoder: Encoder) throws {
        var c = encoder.container(keyedBy: CodingKeys.self)
        switch self {
        case .text(let text): try c.encode(text, forKey: .typedText)
        case .key(let name): try c.encode(name.rawValue, forKey: .keyName)
        }
    }
}

struct ItemsBody: Encodable, Sendable {
    let dialog: String?
    let items: [Item]
    let mark: String?
    let write: String

    enum CodingKeys: String, CodingKey { case dialog, items, mark, write }

    func encode(to encoder: Encoder) throws {
        var c = encoder.container(keyedBy: CodingKeys.self)
        try c.encode(write, forKey: .write)
        try c.encode(dialog, forKey: .dialog)
        try c.encodeIfPresent(mark, forKey: .mark)
        try c.encode(items, forKey: .items)
    }
}

enum WriteRoute: Equatable, Sendable {
    case plain(session: String, batch: Bool)
    case items(dialog: String?, items: [Item])
}

final class DoorClient {
    private func signedPost(route: WriteRoute) -> Data? {
        let body: Data
        do {
            let encoder = JSONEncoder()
            encoder.outputFormatting = [.sortedKeys]
            switch route {
            case .plain(let session, let batch):
                body = try encoder.encode(PlainBody(batch: batch, count: 1, keep: .top, note: nil, session: session))
            case .items(let dialog, let items):
                body = try encoder.encode(ItemsBody(dialog: dialog, items: items, mark: nil, write: "w"))
            }
        } catch {
            return nil
        }
        return body
    }
}

final class Screen {
    private let poll = DoorLine(keeps: true)
    private let typing = DoorLine(keeps: true)
}

final class Pane {
    private let a = DoorLine(keeps: true)
    private let b = DoorLine(keeps: true)
    private let c = DoorLine(keeps: true)
    private let once = DoorLine(keeps: false)

    final class Inner {
        private let d = DoorLine(keeps: true)
    }
}
